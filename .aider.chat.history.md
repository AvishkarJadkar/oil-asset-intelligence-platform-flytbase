import json
from typing import Dict, List, Any

class WellData:
    """
    A container class to hold all structured and unstructured data for a single well.
    """
    def __init__(self, asset_id: str):
        self.asset_id = asset_id
        self.metadata: Dict[str, Any] = {}
        self.inspections: List[Dict[str, Any]] = []
        self.maintenance_reports: List[Dict[str, Any]] = []
        self.regulatory_events: List[Dict[str, Any]] = []
        self.context_notes: List[Dict[str, Any]] = []

    def __repr__(self):
        return f"<WellData for {self.asset_id}>"

def load_json_file(file_path: str) -> Dict[str, Any]:
    """Loads a single JSON file."""
    try:
        with open(file_path, 'r') as f:
            return json.load(f)
    except FileNotFoundError:
        print(f"Error: File not found at {file_path}")
        return {}
    except json.JSONDecodeError:
        print(f"Error: Invalid JSON format in {file_path}")
        return {}

def load_jsonl_file(file_path: str) -> List[Dict[str, Any]]:
    """Loads a JSON Lines (JSONL) file."""
    data = []
    try:
        with open(file_path, 'r') as f:
            for line in f:
                try:
                    data.append(json.loads(line))
                except json.JSONDecodeError:
                    print(f"Skipping invalid JSON line in {file_path}: {line.strip()}")
    except FileNotFoundError:
        print(f"Error: File not found at {file_path}")
    return data

def ingest_well_data(asset_id: str, file_paths: Dict[str, str]) -> WellData:
    """
    Ingests all data sources for a specific well and populates the WellData object.

    Args:
        asset_id: The ID of the well (e.g., "WELL-0001").
        file_paths: A dictionary mapping data type keys to file paths.

    Returns:
        A WellData object containing all parsed information.
    """
    well_data = WellData(asset_id)

    # 1. Load Metadata (Single JSON file)
    if 'metadata' in file_paths:
        meta = load_json_file(file_paths['metadata'])
        well_data.metadata = meta

    # 2. Load Inspections (Single JSONL file)
    if 'inspections' in file_paths:
        inspections = load_jsonl_file(file_paths['inspections'])
        well_data.inspections = inspections

    # 3. Load Maintenance Reports (Single JSONL file)
    if 'maintenance' in file_paths:
        maintenance = load_jsonl_file(file_paths['maintenance'])
        well_data.maintenance_reports = maintenance

    # 4. Load Regulatory Events (Single JSONL file)
    if 'regulatory_events' in file_paths:
        events = load_jsonl_file(file_paths['regulatory_events'])
        well_data.regulatory_events = events

    # 5. Load Context Notes (Single JSONL file)
    if 'context' in file_paths:
        context = load_jsonl_file(file_paths['context'])
        well_data.context_notes = context

    print(f"Successfully ingested data for {asset_id}.")
    return well_data
